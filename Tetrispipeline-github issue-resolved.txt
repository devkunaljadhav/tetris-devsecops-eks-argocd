
Pipeline-1
=================================================================================
pipeline{
    agent any
    tools{
        jdk 'jdk25'
        nodejs 'node25'
    }
    environment {
        SCANNER_HOME=tool 'sonar-scanner'
        GIT_REPO_NAME = "devsecops-Tetris-manifest"
        GIT_USER_NAME = "S4bandi"
    }
    stages {
        stage('clean workspace'){
            steps{
                cleanWs()
            }
        }
        stage('Checkout from Git'){
            steps{
                git branch: 'main', url: 'https://github.com/devkunaljadhav/devsecops-tetris-v2.git'
            }
        }
        stage("Sonarqube Analysis "){
            steps{
                withSonarQubeEnv('sonar-server') {
                    sh ''' $SCANNER_HOME/bin/sonar-scanner -Dsonar.projectName=TetrisVersion2.0 \
                    -Dsonar.projectKey=TetrisVersion2.0 '''
                }
            }
        }
        stage("Quality Gate"){
           steps {
                script {
                    waitForQualityGate abortPipeline: false, credentialsId: 'sonar-token'
                }
            }
        }
        stage('Install Dependencies') {
            steps {
                sh "npm install"
            }
        }
        stage('TRIVY FS SCAN') {
            steps {
                sh "trivy fs . > trivyfs.txt"
            }
        }
        stage("Docker Build & Push"){
            steps{
                script{
                   withDockerRegistry(credentialsId: 'docker', toolName: 'docker'){
                       sh "docker build -t tetrisv2 ."
                       sh "docker tag tetrisv2 kunaljadhav1625/tetrisv2:{BUILD_NUMBER} "
                       sh "docker push kunaljadhav1625/tetrisv2:{BUILD_NUMBER} "
                    }
                }
            }
        }
        stage("TRIVY"){
            steps{
                sh "trivy image kunaljadhav1625/tetrisv2:{BUILD_NUMBER} > trivyimage.txt"
            }
        }
        stage('Checkout Code') {
            steps {
                git branch: 'main', url: 'https://github.com/devkunaljadhav/devsecops-Tetris-manifest.git'
            }
        }
        stage('Update Deployment File') {
            environment {
                NEW_IMAGE_NAME = "S4bandi/tetrisv1:${BUILD_NUMBER}"
        }
        steps {
            withCredentials([string(credentialsId: 'github', variable: 'GITHUB_TOKEN')]) {
                sh '''
                   sed -i "s|image: .*|image: ${NEW_IMAGE_NAME}|" deployment.yml
                   git config user.email "kunaljadhav1625@gmail.com"
                   git config user.name "kunal"
                   git add deployment.yml
                   git diff --cached --quiet || git commit -m "Update deployment image to ${NEW_IMAGE_NAME}"
                   git push https://${GIT_USER_NAME}:${GITHUB_TOKEN}@github.com/${GIT_USER_NAME}/${GIT_REPO_NAME}.git HEAD:main
                '''
                }
            }
        }
        
    }
}
